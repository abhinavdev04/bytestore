<?php
// Shared product handler - processes add/update for admin and employee
function handleProductAction($conn, $user_type, $user_id) {
    $success = '';
    $error = '';
    
    if (isset($_POST['add_product'])) {
        $name = mysqli_real_escape_string($conn, $_POST['product_name']);
        $description = mysqli_real_escape_string($conn, $_POST['product_description']);
        $price = mysqli_real_escape_string($conn, $_POST['product_price']);
        $stock = (int)$_POST['product_stock'];
        $image_path = mysqli_real_escape_string($conn, $_POST['product_image_path']);
        
        $sql = "INSERT INTO product (product_name, product_description, product_price, product_image_path, product_stock) 
                VALUES ('$name', '$description', $price, '$image_path', $stock)";
        
        if (mysqli_query($conn, $sql)) {
            $product_id = mysqli_insert_id($conn);
            logProductAction($conn, $user_type, $user_id, $product_id, 'ADD');
            $success = "Product added successfully!";
        } else {
            $error = "Failed to add product.";
        }
    }
    
    if (isset($_POST['update_product'])) {
        $product_id = (int)$_POST['product_id'];
        $name = mysqli_real_escape_string($conn, $_POST['product_name']);
        $description = mysqli_real_escape_string($conn, $_POST['product_description']);
        $price = mysqli_real_escape_string($conn, $_POST['product_price']);
        $stock = (int)$_POST['product_stock'];
        $image_path = mysqli_real_escape_string($conn, $_POST['product_image_path']);
        
        $update = "UPDATE product SET product_name='$name', product_description='$description', 
                   product_price=$price, product_stock=$stock, product_image_path='$image_path' 
                   WHERE product_id=$product_id";
        
        if (mysqli_query($conn, $update)) {
            logProductAction($conn, $user_type, $user_id, $product_id, 'UPDATE');
            $success = "Product updated successfully!";
        } else {
            $error = "Failed to update product.";
        }
    }
    
    return ['success' => $success, 'error' => $error];
}

function logProductAction($conn, $user_type, $user_id, $product_id, $action) {
    if ($user_type == 'admin') {
        $log = "INSERT INTO admin_manages_product (admin_id, product_id, action_type) VALUES ($user_id, $product_id, '$action')";
    } else {
        $log = "INSERT INTO employee_manages_product (employee_id, product_id, action_type) VALUES ($user_id, $product_id, '$action')";
    }
    mysqli_query($conn, $log);
}
?>
